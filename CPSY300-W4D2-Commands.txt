name: Show my subject

on: workflow_dispatch

jobs:
  whoami:
    runs-on: ubuntu-latest
    steps:
      - name: Print both possible subjects
        env:
          OWNER: ${{ github.repository_owner }}
          OWNER_ID: ${{ github.repository_owner_id }}
          FULL: ${{ github.repository }}
          REPO_ID: ${{ github.repository_id }}
          REF: ${{ github.ref }}
        run: |
          REPO="${FULL##*/}"
          echo "NAME: repo:${FULL}:ref:${REF}"
          echo "IDS:  repo:${OWNER}@${OWNER_ID}/${REPO}@${REPO_ID}:ref:${REF}"

Then PASTE the rest of the block:

# Create the resource group
az group create -n $RG -l $LOC
# Create the registry the pipeline will push images to (Basic tier)
az acr create -n $ACR -g $RG --sku Basic --admin-enabled true
# Create the pipeline's own identity, gh-deploy: an identity with no password
az identity create -n gh-deploy -g $RG -l $LOC

# --- part 2: read back four values and keep them in variables (nothing is printed)
# CLIENT_ID = the identity's ID that GitHub will sign in with
CLIENT_ID=$(az identity show -n gh-deploy -g $RG --query clientId -o tsv)
# PRINCIPAL = the identity's ID that permissions are given to
PRINCIPAL=$(az identity show -n gh-deploy -g $RG --query principalId -o tsv)
# SUB = the ID of your Azure subscription
SUB=$(az account show --query id -o tsv)
# TENANT = the ID of your directory (SAIT's)
TENANT=$(az account show --query tenantId -o tsv)

BLOCK 8 · github.com · file name: .github/workflows/deploy.yml
After pasting, change the three lines under env: to YOUR names, exactly as
you created them in Cloud Shell. Change nothing else.
---------------------------------------------------------------------------
name: Build and deploy

on:
  push:
    branches: [ main ]
  workflow_dispatch:

permissions:
  id-token: write
  contents: read

env:
  ACR_NAME: acr<yoursurname><4 digits>
  RESOURCE_GROUP: rg-cpsy300-w4d2-<your initials>
  CONTAINERAPP: ca-pipeline-<your initials>

jobs:
  build-and-deploy:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v7

      - uses: azure/login@v3
        with:
          client-id: ${{ secrets.AZURE_CLIENT_ID }}
          tenant-id: ${{ secrets.AZURE_TENANT_ID }}
          subscription-id: ${{ secrets.AZURE_SUBSCRIPTION_ID }}

      - name: Sign in to the registry
        run: az acr login --name $ACR_NAME

      - name: Build and push
        run: |
          IMAGE=$ACR_NAME.azurecr.io/pipeline:${{ github.sha }}
          docker build -t $IMAGE .
          docker push $IMAGE

      - name: Deploy
        run: |
          az containerapp update \
            --name $CONTAINERAPP \
            --resource-group $RESOURCE_GROUP \
            --image $ACR_NAME.azurecr.io/pipeline:${{ github.sha }}
